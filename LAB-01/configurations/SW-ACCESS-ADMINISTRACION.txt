! ║  SW-ACCESS-ADMINISTRACIÓN (VLAN 30 y VLAN 40)                  ║
! ╠════════════════════════════════════════════════════════════════╣
! ║                                                                ║
! ║  Este switch atiende dos áreas:                                ║
! ║    • ADMINISTRACIÓN  →  VLAN 30  →  g0/2-3                     ║
! ║    • GERENCIA        →  VLAN 40  →  g1/0-1                     ║
! ║                                                                ║
! ║  Ambas comparten la misma base (hostname, VLANs, EtherChannel, ║
! ║  trunk, SVI de gestión y SSH), por eso están en un solo        ║
! ║  archivo. Cada área tiene su propia sección de puertos de      ║
! ║  acceso para que sea fácil ubicarlas.                          ║
! ║                                                                ║
! ╚════════════════════════════════════════════════════════════════╝
  ============================================================

! ============================================================
! CONFIGURACION-SW-ACCESS-ADMINISTRACION
! Proyecto: Laboratorio personal de redes y ciberseguridad
! Plataforma: Cisco IOS en GNS3 (entorno de práctica, no producción)
! Áreas: ADMINISTRACIÓN (VLAN 30) y GERENCIA (VLAN 40)
! ============================================================

ACCESO-Y-MODO-GLOBAL
---------------------------------------
> enable
# configure terminal
(config)#

IDENTIDAD-Y-SEGURIDAD-BASICA
---------------------------------------
(config)# hostname SW-ACCESS-ADMINISTRACION
(config)# ip domain-name lab-local
(config)# no ip domain-lookup
(config)# service password-encryption
(config)# enable secret admin
(config)# banner motd # ACCESO AUTORIZADO UNICAMENTE - PyS Networks #

LINEA-CONSOLA
---------------------------------------
(config)# line console 0
(config-line)# password admin
(config-line)# login
(config-line)# logging synchronous
(config-line)# exec-timeout 5 0
(config-line)# exit

CREACION-DE-VLANS
---------------------------------------
(config)# vlan 10
(config-vlan)# name DESARROLLO
(config-vlan)# exit
(config)# vlan 20
(config-vlan)# name SOPORTE-TI
(config-vlan)# exit
(config)# vlan 30
(config-vlan)# name ADMIN
(config-vlan)# exit
(config)# vlan 40
(config-vlan)# name GERENCIA
(config-vlan)# exit
(config)# vlan 99
(config-vlan)# name GESTION
(config-vlan)# exit
(config)# vlan 998
(config-vlan)# name BLACK-HOLE
(config-vlan)# exit
(config)# vlan 999
(config-vlan)# name NATIVA
(config-vlan)# exit

ETHERCHANNEL-03-LACP
---------------------------------------
(config)# interface port-channel 3
(config-if)# description LACP-HACIA-PyS-SW-CORE
(config-if)# switchport trunk encapsulation dot1q
(config-if)# switchport mode trunk
(config-if)# switchport trunk native vlan 999
(config-if)# switchport trunk allowed vlan 10,20,30,40,99,999
(config-if)# no shutdown
(config-if)# exit
(config)# interface range g0/0-1
(config-if-range)# description MIEMBROS-LACP-PO3
(config-if-range)# channel-group 3 mode active
(config-if-range)# no shutdown
(config-if-range)# exit

PUERTOS-DE-ACCESO-ADMINISTRACION
---------------------------------------
(config)# interface range g0/2-3
(config-if-range)# description PUERTOS-ACCESO-ADMINISTRACION
(config-if-range)# switchport mode access
(config-if-range)# switchport access vlan 30
(config-if-range)# no shutdown
(config-if-range)# exit

PUERTOS-DE-ACCESO-GERENCIA
---------------------------------------
(config)# interface range g1/0-1
(config-if-range)# description PUERTOS-ACCESO-GERENCIA
(config-if-range)# switchport mode access
(config-if-range)# switchport access vlan 40
(config-if-range)# no shutdown
(config-if-range)# exit

PUERTOS-SIN-USO-BLACK-HOLE
---------------------------------------
(config)# interface range g1/2-3,g2/0-3,g3/0-3
(config-if-range)# description PUERTOS-SIN-USO
(config-if-range)# switchport mode access
(config-if-range)# switchport access vlan 998
(config-if-range)# shutdown
(config-if-range)# exit

SVI-GESTION-VLAN-99
---------------------------------------
(config)# interface vlan 99
(config-if)# description GESTION-SW-ACCESS-ADMINISTRACION
(config-if)# ip address 192.168.10.125 255.255.255.248
(config-if)# no shutdown
(config-if)# exit
(config)# ip default-gateway 192.168.10.121

SSHv2-ADMINISTRACION-REMOTA
---------------------------------------
(config)# crypto key generate rsa modulus 2048
(config)# username admin privilege 15 secret admin
(config)# ip ssh version 2
(config)# line vty 0 15
(config-line)# login local
(config-line)# transport input ssh
(config-line)# logging synchronous
(config-line)# exec-timeout 10 0
(config-line)# exit

GUARDAR-CONFIGURACION
---------------------------------------
(config)# end
# write memory
