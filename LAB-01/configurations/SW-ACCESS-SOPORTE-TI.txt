ACCESO-Y-MODO-GLOBAL
---------------------------------------
> enable
# configure terminal
(config)#

IDENTIDAD-Y-SEGURIDAD-BASICA
---------------------------------------
(config)# hostname SW-ACCESS-SOPORTE-TI
(config)# ip domain-name lab-local
(config)# no ip domain-lookup
(config)# service password-encryption
(config)# enable secret admin
(config)# banner motd # ACCESO AUTORIZADO UNICAMENTE - PyS Networks #

LINEA-CONSOLA
---------------------------------------
(config)# line console 0
(config-line)# password admin
(config-line)# login
(config-line)# logging synchronous
(config-line)# exec-timeout 10 0
(config-line)# exit

ETHERCHANNEL-02-LACP
---------------------------------------
(config)# interface port-channel 2
(config-if)# description LACP-HACIA-SW-CORE
(config-if)# switchport trunk encapsulation dot1q
(config-if)# switchport mode trunk
(config-if)# switchport trunk native vlan 999
(config-if)# switchport trunk allowed vlan 10,20,30,40,99,999
(config-if)# no shutdown
(config-if)# exit
(config)# interface range g0/0-1
(config-if-range)# description MIEMBROS-LACP-PO2
(config-if-range)# channel-group 2 mode active
(config-if-range)# no shutdown
(config-if-range)# exit

---------------------------------------
