
CONFIGURACION-R1
-----------------------
enable
configure terminal
hostname PyS-R1
no ip domain-lookup
ip domain-name lab-local
service password-encryption
enable secret admin
banner motd # ACCESO AUTORIZADO UNICAMENTE - PyS Networks #
!
line console 0
 password admin
 login
 logging synchronous
 exec-timeout 10 0
exit
!
